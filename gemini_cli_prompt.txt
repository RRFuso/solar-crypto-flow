"""
Sistema de Detecção de Oportunidades de Trade
Implementa algoritmos para identificar pontos de entrada baseados em:
- Suportes e resistências
- Pullbacks após rompimentos
- Regiões de sobrevenda com volume
- Cruzamentos de médias móveis
- Inflow/Outflow de capital
"""

import numpy as np
import pandas as pd
from typing import Dict, List, Tuple, Optional
from dataclasses import dataclass
from datetime import datetime, timedelta
import talib

@dataclass
class TradeOpportunity:
    """Representa uma oportunidade de trade detectada"""
    symbol: str
    signal_type: str
    direction: str  # 'long' or 'short'
    entry_price: float
    confidence: float
    stop_loss: float
    take_profit_targets: List[float]
    risk_reward_ratio: float
    volume_confirmation: bool
    timestamp: datetime
    reasons: List[str]
    leverage_recommendation: float

class TechnicalIndicators:
    """Classe para cálculo de indicadores técnicos"""
    
    @staticmethod
    def calculate_support_resistance(df: pd.DataFrame, window: int = 20) -> Tuple[List[float], List[float]]:
        """Identifica níveis de suporte e resistência"""
        highs = df['high'].rolling(window=window, center=True).max()
        lows = df['low'].rolling(window=window, center=True).min()
        
        # Encontrar picos e vales
        resistance_levels = []
        support_levels = []
        
        for i in range(window, len(df) - window):
            if df['high'].iloc[i] == highs.iloc[i]:
                resistance_levels.append(df['high'].iloc[i])
            if df['low'].iloc[i] == lows.iloc[i]:
                support_levels.append(df['low'].iloc[i])
        
        # Remover níveis muito próximos
        resistance_levels = TechnicalIndicators._filter_close_levels(resistance_levels)
        support_levels = TechnicalIndicators._filter_close_levels(support_levels)
        
        return support_levels, resistance_levels
    
    @staticmethod
    def _filter_close_levels(levels: List[float], threshold: float = 0.02) -> List[float]:
        """Remove níveis muito próximos entre si"""
        if not levels:
            return []
        
        levels = sorted(levels)
        filtered = [levels[0]]
        
        for level in levels[1:]:
            if abs(level - filtered[-1]) / filtered[-1] > threshold:
                filtered.append(level)
        
        return filtered
    
    @staticmethod
    def detect_pullback(df: pd.DataFrame, breakout_threshold: float = 0.02) -> Dict:
        """Detecta pullbacks após rompimentos"""
        support_levels, resistance_levels = TechnicalIndicators.calculate_support_resistance(df)
        current_price = df['close'].iloc[-1]
        
        pullback_signals = []
        
        # Verificar rompimento de resistência seguido de pullback
        for resistance in resistance_levels:
            if current_price > resistance * (1 + breakout_threshold):
                # Verificar se houve pullback para a resistência
                recent_low = df['low'].tail(5).min()
                if recent_low <= resistance * 1.01 and recent_low >= resistance * 0.99:
                    pullback_signals.append({
                        'type': 'bullish_pullback',
                        'level': resistance,
                        'current_price': current_price,
                        'confidence': 0.8
                    })
        
        # Verificar rompimento de suporte seguido de pullback
        for support in support_levels:
            if current_price < support * (1 - breakout_threshold):
                # Verificar se houve pullback para o suporte
                recent_high = df['high'].tail(5).max()
                if recent_high >= support * 0.99 and recent_high <= support * 1.01:
                    pullback_signals.append({
                        'type': 'bearish_pullback',
                        'level': support,
                        'current_price': current_price,
                        'confidence': 0.8
                    })
        
        return pullback_signals
    
    @staticmethod
    def detect_oversold_with_volume(df: pd.DataFrame, rsi_threshold: float = 30, volume_multiplier: float = 1.5) -> Dict:
        """Detecta regiões de sobrevenda com confirmação de volume"""
        # Calcular RSI
        rsi = talib.RSI(df['close'].values, timeperiod=14)
        current_rsi = rsi[-1]
        
        # Calcular volume médio
        avg_volume = df['volume'].rolling(window=20).mean()
        current_volume = df['volume'].iloc[-1]
        avg_volume_20 = avg_volume.iloc[-1]
        
        # Verificar condições
        is_oversold = current_rsi < rsi_threshold
        high_volume = current_volume > avg_volume_20 * volume_multiplier
        
        if is_oversold and high_volume:
            return {
                'signal': True,
                'rsi': current_rsi,
                'volume_ratio': current_volume / avg_volume_20,
                'confidence': min(0.9, (rsi_threshold - current_rsi) / rsi_threshold + \
                                (current_volume / avg_volume_20 - 1) * 0.2)
            }
        
        return {'signal': False}
    
    @staticmethod
    def detect_ma_crossover(df: pd.DataFrame, fast_period: int = 9, slow_period: int = 21) -> Dict:
        """Detecta cruzamentos de médias móveis"""
        # Calcular médias móveis
        ma_fast = talib.SMA(df['close'].values, timeperiod=fast_period)
        ma_slow = talib.SMA(df['close'].values, timeperiod=slow_period)
        
        # Verificar cruzamentos
        current_fast = ma_fast[-1]
        current_slow = ma_slow[-1]
        prev_fast = ma_fast[-2]
        prev_slow = ma_slow[-2]
        
        signals = []
        
        # Golden Cross (bullish)
        if prev_fast <= prev_slow and current_fast > current_slow:
            signals.append({
                'type': 'golden_cross',
                'direction': 'bullish',
                'fast_ma': current_fast,
                'slow_ma': current_slow,
                'confidence': 0.7
            })
        
        # Death Cross (bearish)
        elif prev_fast >= prev_slow and current_fast < current_slow:
            signals.append({
                'type': 'death_cross',
                'direction': 'bearish',
                'fast_ma': current_fast,
                'slow_ma': current_slow,
                'confidence': 0.7
            })
        
        return signals
    
    @staticmethod
    def analyze_capital_flow(df: pd.DataFrame) -> Dict:
        """Analisa inflow/outflow de capital usando OBV e Money Flow Index"""
        # On-Balance Volume
        obv = talib.OBV(df['close'].values, df['volume'].values)
        obv_sma = talib.SMA(obv, timeperiod=10)
        
        # Money Flow Index
        mfi = talib.MFI(df['high'].values, df['low'].values, df['close'].values, df['volume'].values, timeperiod=14)
        
        current_obv = obv[-1]
        current_obv_sma = obv_sma[-1]
        current_mfi = mfi[-1]
        
        # Determinar fluxo
        obv_trend = 'inflow' if current_obv > current_obv_sma else 'outflow'
        mfi_signal = 'oversold' if current_mfi < 20 else 'overbought' if current_mfi > 80 else 'neutral'
        
        return {
            'obv_trend': obv_trend,
            'mfi': current_mfi,
            'mfi_signal': mfi_signal,
            'confidence': abs(current_mfi - 50) / 50  # Quanto mais extremo, maior a confiança
        }

class TradeOpportunityDetector:
    """Detector principal de oportunidades de trade"""
    
    def __init__(self):
        self.indicators = TechnicalIndicators()
    
    def analyze_symbol(self, df: pd.DataFrame, symbol: str) -> List[TradeOpportunity]:
        """Analisa um símbolo e retorna oportunidades detectadas"""
        opportunities = []
        current_price = df['close'].iloc[-1]
        
        # 1. Detectar pullbacks
        pullback_signals = self.indicators.detect_pullback(df)
        for signal in pullback_signals:
            opportunity = self._create_pullback_opportunity(signal, symbol, current_price)
            if opportunity:
                opportunities.append(opportunity)
        
        # 2. Detectar sobrevenda com volume
        oversold_signal = self.indicators.detect_oversold_with_volume(df)
        if oversold_signal['signal']:
            opportunity = self._create_oversold_opportunity(oversold_signal, symbol, current_price)
            opportunities.append(opportunity)
        
        # 3. Detectar cruzamentos de médias
        ma_signals = self.indicators.detect_ma_crossover(df)
        for signal in ma_signals:
            opportunity = self._create_ma_crossover_opportunity(signal, symbol, current_price)
            opportunities.append(opportunity)
        
        # 4. Analisar fluxo de capital
        capital_flow = self.indicators.analyze_capital_flow(df)
        if capital_flow['confidence'] > 0.6:
            opportunity = self._create_capital_flow_opportunity(capital_flow, symbol, current_price)
            if opportunity:
                opportunities.append(opportunity)
        
        return opportunities
    
    def _create_pullback_opportunity(self, signal: Dict, symbol: str, current_price: float) -> Optional[TradeOpportunity]:
        """Cria oportunidade baseada em pullback"""
        if signal['type'] == 'bullish_pullback':
            stop_loss = signal['level'] * 0.98
            take_profit_targets = [
                current_price * 1.02,
                current_price * 1.05,
                current_price * 1.08
            ]
            direction = 'long'
            reasons = [f"Pullback bullish para resistência em {signal['level']:.4f}"]
        else:
            stop_loss = signal['level'] * 1.02
            take_profit_targets = [
                current_price * 0.98,
                current_price * 0.95,
                current_price * 0.92
            ]
            direction = 'short'
            reasons = [f"Pullback bearish para suporte em {signal['level']:.4f}"]
        
        risk_reward = abs(take_profit_targets[0] - current_price) / abs(stop_loss - current_price)
        
        return TradeOpportunity(
            symbol=symbol,
            signal_type='pullback',
            direction=direction,
            entry_price=current_price,
            confidence=signal['confidence'],
            stop_loss=stop_loss,
            take_profit_targets=take_profit_targets,
            risk_reward_ratio=risk_reward,
            volume_confirmation=True,
            timestamp=datetime.now(),
            reasons=reasons,
            leverage_recommendation=min(3.0, risk_reward)
        )
    
    def _create_oversold_opportunity(self, signal: Dict, symbol: str, current_price: float) -> TradeOpportunity:
        """Cria oportunidade baseada em sobrevenda"""
        stop_loss = current_price * 0.95
        take_profit_targets = [
            current_price * 1.03,
            current_price * 1.06,
            current_price * 1.10
        ]
        
        risk_reward = abs(take_profit_targets[0] - current_price) / abs(stop_loss - current_price)
        
        return TradeOpportunity(
            symbol=symbol,
            signal_type='oversold_volume',
            direction='long',
            entry_price=current_price,
            confidence=signal['confidence'],
            stop_loss=stop_loss,
            take_profit_targets=take_profit_targets,
            risk_reward_ratio=risk_reward,
            volume_confirmation=True,
            timestamp=datetime.now(),
            reasons=[
                f"RSI em sobrevenda: {signal['rsi']:.2f}",
                f"Volume elevado: {signal['volume_ratio']:.2f}x da média"
            ],
            leverage_recommendation=min(2.0, risk_reward)
        )
    
    def _create_ma_crossover_opportunity(self, signal: Dict, symbol: str, current_price: float) -> TradeOpportunity:
        """Cria oportunidade baseada em cruzamento de médias"""
        if signal['direction'] == 'bullish':
            stop_loss = current_price * 0.97
            take_profit_targets = [
                current_price * 1.02,
                current_price * 1.04,
                current_price * 1.07
            ]
            direction = 'long'
        else:
            stop_loss = current_price * 1.03
            take_profit_targets = [
                current_price * 0.98,
                current_price * 0.96,
                current_price * 0.93
            ]
            direction = 'short'
        
        risk_reward = abs(take_profit_targets[0] - current_price) / abs(stop_loss - current_price)
        
        return TradeOpportunity(
            symbol=symbol,
            signal_type='ma_crossover',
            direction=direction,
            entry_price=current_price,
            confidence=signal['confidence'],
            stop_loss=stop_loss,
            take_profit_targets=take_profit_targets,
            risk_reward_ratio=risk_reward,
            volume_confirmation=False,

        """Cria oportunidade baseada em fluxo de capital"""

"""
Sistema de Gestão de Risco e Pontos de Realização
Calcula e gerencia risco/retorno, alavancagem e pontos de realização para trades
"""

import math
from typing import Dict, List, Tuple, Optional
from dataclasses import dataclass
from datetime import datetime
import numpy as np

@dataclass
class RiskMetrics:
    """Métricas de risco para um trade"""
    position_size: float
    risk_amount: float
    risk_percentage: float
    reward_amount: float
    risk_reward_ratio: float
    max_leverage: float
    recommended_leverage: float
    stop_loss_distance: float
    take_profit_distances: List[float]

@dataclass
class RealizationPoint:
    """Ponto de realização de lucro"""
    level: int  # 1, 2, 3, etc.
    price: float
    percentage_to_close: float  # Porcentagem da posição a fechar
    profit_percentage: float  # Porcentagem de lucro neste nível
    hit: bool = False
    hit_time: Optional[datetime] = None

class RiskCalculator:
    """Calculadora de risco e gestão de posições"""
    
    def __init__(self, account_balance: float = 10000.0):
        self.account_balance = account_balance
        self.max_risk_per_trade = 0.02  # 2% por trade
        self.max_daily_risk = 0.10  # 10% por dia
        self.max_leverage = 10.0
    
    def calculate_position_size(self, entry_price: float, stop_loss: float, \
                              risk_percentage: float = None) -> Dict:
        """Calcula o tamanho da posição baseado no risco"""
        if risk_percentage is None:
            risk_percentage = self.max_risk_per_trade
        
        risk_amount = self.account_balance * risk_percentage
        stop_loss_distance = abs(entry_price - stop_loss)
        stop_loss_percentage = stop_loss_distance / entry_price
        
        # Tamanho da posição sem alavancagem
        base_position_size = risk_amount / stop_loss_distance
        
        # Valor notional da posição
        notional_value = base_position_size * entry_price
        
        # Calcular alavancagem necessária
        required_margin = notional_value / self.max_leverage
        available_margin = self.account_balance * 0.8  # Usar 80% do saldo
        
        if required_margin > available_margin:
            # Reduzir posição para caber na margem disponível
            max_notional = available_margin * self.max_leverage
            base_position_size = max_notional / entry_price
            notional_value = max_notional
        
        return {
            'position_size': base_position_size,
            'notional_value': notional_value,
            'risk_amount': risk_amount,
            'stop_loss_distance': stop_loss_distance,
            'stop_loss_percentage': stop_loss_percentage,
            'required_margin': required_margin
        }
    
    def calculate_optimal_leverage(self, entry_price: float, stop_loss: float, \
                                 target_risk: float = None) -> float:
        """Calcula a alavancagem ótima baseada no risco"""
        if target_risk is None:
            target_risk = self.max_risk_per_trade
        
        stop_loss_distance = abs(entry_price - stop_loss)
        stop_loss_percentage = stop_loss_distance / entry_price
        
        # Alavancagem necessária para atingir o risco desejado
        optimal_leverage = target_risk / stop_loss_percentage
        
        # Limitar à alavancagem máxima
        return min(optimal_leverage, self.max_leverage)
    
    def calculate_risk_reward_ratio(self, entry_price: float, stop_loss: float, \
                                  take_profit: float) -> float:
        """Calcula a relação risco/retorno"""
        risk = abs(entry_price - stop_loss)
        reward = abs(take_profit - entry_price)
        
        if risk == 0:
            return 0
        
        return reward / risk
    
    def generate_take_profit_levels(self, entry_price: float, direction: str, \
                                  stop_loss: float, num_levels: int = 3) -> List[RealizationPoint]:
        """Gera níveis de take profit baseados em Fibonacci e suporte/resistência"""
        
        stop_distance = abs(entry_price - stop_loss)
        
        # Ratios de Fibonacci para take profits
        fibonacci_ratios = [1.618, 2.618, 4.236, 6.854]
        
        # Porcentagens da posição a fechar em cada nível
        close_percentages = [30, 40, 30]  # 30%, 40%, 30%
        
        if num_levels == 4:
            close_percentages = [25, 25, 25, 25]
        elif num_levels == 2:
            close_percentages = [50, 50]
        
        take_profit_levels = []
        
        for i in range(min(num_levels, len(fibonacci_ratios))):
            if direction == 'long':
                tp_price = entry_price + (stop_distance * fibonacci_ratios[i])
                profit_percentage = ((tp_price - entry_price) / entry_price) * 100
            else:  # short
                tp_price = entry_price - (stop_distance * fibonacci_ratios[i])
                profit_percentage = ((entry_price - tp_price) / entry_price) * 100
            
            take_profit_levels.append(RealizationPoint(
                level=i + 1,
                price=tp_price,
                percentage_to_close=close_percentages[i] if i < len(close_percentages) else 100,
                profit_percentage=profit_percentage
            ))
        
        return take_profit_levels
    
    def calculate_comprehensive_risk_metrics(self, entry_price: float, stop_loss: float, \
                                           take_profit_levels: List[float], \
                                           direction: str) -> RiskMetrics:
        """Calcula métricas completas de risco"""
        
        position_data = self.calculate_position_size(entry_price, stop_loss)
        optimal_leverage = self.calculate_optimal_leverage(entry_price, stop_loss)
        
        # Calcular distâncias
        stop_loss_distance = abs(entry_price - stop_loss) / entry_price * 100
        take_profit_distances = []
        
        for tp in take_profit_levels:
            distance = abs(tp - entry_price) / entry_price * 100
            take_profit_distances.append(distance)
        
        # Calcular reward médio ponderado
        total_reward = 0
        for tp in take_profit_levels:
            if direction == 'long':
                reward = tp - entry_price
            else:
                reward = entry_price - tp
            total_reward += reward
        
        avg_reward = total_reward / len(take_profit_levels)
        risk_amount = abs(entry_price - stop_loss)
        risk_reward_ratio = avg_reward / risk_amount if risk_amount > 0 else 0
        
        return RiskMetrics(
            position_size=position_data['position_size'],
            risk_amount=position_data['risk_amount'],
            risk_percentage=self.max_risk_per_trade * 100,
            reward_amount=avg_reward * position_data['position_size'],
            risk_reward_ratio=risk_reward_ratio,
            max_leverage=self.max_leverage,
            recommended_leverage=optimal_leverage,
            stop_loss_distance=stop_loss_distance,
            take_profit_distances=take_profit_distances
        )

class PositionManager:
    """Gerenciador de posições e realização de lucros"""
    
    def __init__(self):
        self.active_positions = {}
        self.realized_profits = {}
    
    def add_position(self, trade_id: str, entry_price: float, quantity: float, \
                    direction: str, take_profit_levels: List[RealizationPoint]):
        """Adiciona uma nova posição para monitoramento"""
        self.active_positions[trade_id] = {
            'entry_price': entry_price,
            'quantity': quantity,
            'direction': direction,
            'take_profit_levels': take_profit_levels,
            'remaining_quantity': quantity,
            'realized_pnl': 0.0,
            'unrealized_pnl': 0.0
        }
    
    def check_take_profit_hits(self, trade_id: str, current_price: float) -> List[RealizationPoint]:
        """Verifica se algum nível de take profit foi atingido"""
        if trade_id not in self.active_positions:
            return []
        
        position = self.active_positions[trade_id]
        direction = position['direction']
        hits = []
        
        for tp_level in position['take_profit_levels']:
            if tp_level.hit:
                continue
            
            hit = False
            if direction == 'long' and current_price >= tp_level.price:
                hit = True
            elif direction == 'short' and current_price <= tp_level.price:
                hit = True
            
            if hit:
                tp_level.hit = True
                tp_level.hit_time = datetime.now()
                hits.append(tp_level)
                
                # Calcular realização parcial
                quantity_to_close = position['remaining_quantity'] * (tp_level.percentage_to_close / 100)
                position['remaining_quantity'] -= quantity_to_close
                
                # Calcular PnL realizado
                if direction == 'long':
                    pnl = (tp_level.price - position['entry_price']) * quantity_to_close
                else:
                    pnl = (position['entry_price'] - tp_level.price) * quantity_to_close
                
                position['realized_pnl'] += pnl
        
        return hits
    
    def calculate_unrealized_pnl(self, trade_id: str, current_price: float) -> float:
        """Calcula o PnL não realizado da posição"""
        if trade_id not in self.active_positions:
            return 0.0
        
        position = self.active_positions[trade_id]
        remaining_qty = position['remaining_quantity']
        entry_price = position['entry_price']
        direction = position['direction']
        
        if direction == 'long':
            unrealized_pnl = (current_price - entry_price) * remaining_qty
        else:
            unrealized_pnl = (entry_price - current_price) * remaining_qty
        
        position['unrealized_pnl'] = unrealized_pnl
        return unrealized_pnl
    
    def get_position_summary(self, trade_id: str, current_price: float) -> Dict:
        """Retorna um resumo completo da posição"""
        if trade_id not in self.active_positions:
            return {}
        
        position = self.active_positions[trade_id]
        unrealized_pnl = self.calculate_unrealized_pnl(trade_id, current_price)
        
        total_pnl = position['realized_pnl'] + unrealized_pnl
        total_pnl_percentage = (total_pnl / (position['entry_price'] * position['quantity'])) * 100
        
        # Contar targets atingidos
        targets_hit = sum(1 for tp in position['take_profit_levels'] if tp.hit)
        total_targets = len(position['take_profit_levels'])
        
        return {
            'trade_id': trade_id,
            'entry_price': position['entry_price'],
            'current_price': current_price,
            'direction': position['direction'],
            'original_quantity': position['quantity'],
            'remaining_quantity': position['remaining_quantity'],
            'realized_pnl': position['realized_pnl'],
            'unrealized_pnl': unrealized_pnl,
            'total_pnl': total_pnl,
            'total_pnl_percentage': total_pnl_percentage,
            'targets_hit': targets_hit,
            'total_targets': total_targets,
            'completion_percentage': (targets_hit / total_targets) * 100
        }

class LeverageOptimizer:
    """Otimizador de alavancagem baseado em volatilidade e risco"""
    
    def __init__(self):
        self.volatility_window = 20
        self.max_leverage = 10.0
        self.conservative_multiplier = 0.7
    
    def calculate_volatility_adjusted_leverage(self, price_history: List[float], \
                                             base_leverage: float) -> float:
        """Ajusta a alavancagem baseada na volatilidade histórica"""
        if len(price_history) < self.volatility_window:
            return base_leverage * self.conservative_multiplier
        
        # Calcular volatilidade (desvio padrão dos retornos)
        returns = []
        for i in range(1, len(price_history)):
            ret = (price_history[i] - price_history[i-1]) / price_history[i-1]
            returns.append(ret)
        
        volatility = np.std(returns) * 100  # Em porcentagem
        
        # Ajustar alavancagem baseada na volatilidade
        # Maior volatilidade = menor alavancagem
        if volatility > 5.0:  # Alta volatilidade
            adjustment_factor = 0.5
        elif volatility > 3.0:  # Média volatilidade
            adjustment_factor = 0.7
        elif volatility > 1.0:  # Baixa volatilidade
            adjustment_factor = 0.9
        else:  # Muito baixa volatilidade
            adjustment_factor = 1.0
        
        adjusted_leverage = base_leverage * adjustment_factor
        return min(adjusted_leverage, self.max_leverage)
    
    def calculate_kelly_criterion_leverage(self, win_rate: float, avg_win: float, \
                                         avg_loss: float) -> float:
        """Calcula alavancagem ótima usando o Critério de Kelly"""
        if avg_loss == 0 or win_rate == 0:
            return 1.0
        
        # Fórmula de Kelly: f = (bp - q) / b
        # onde b = avg_win/avg_loss, p = win_rate, q = 1 - win_rate
        b = avg_win / abs(avg_loss)
        p = win_rate
        q = 1 - win_rate
        
        kelly_fraction = (b * p - q) / b
        
        # Aplicar fração conservadora (25% do Kelly)
        conservative_kelly = kelly_fraction * 0.25
        
        # Converter para alavancagem (limitada)
        leverage = max(1.0, min(conservative_kelly * 10, self.max_leverage))
        
        return leverage

        # Gerar níveis de take profit otimizados
        tp_levels = self.risk_calculator.generate_take_profit_levels(
            opportunity.entry_price, \
            opportunity.direction, \
            opportunity.stop_loss, \
            num_levels=3
        )
        
        # Atualizar targets de take profit
        opportunity.take_profit_targets = [tp.price for tp in tp_levels]
        
        # Calcular alavancagem otimizada baseada na volatilidade
        price_history = price_data['close'].tail(20).tolist()
        optimized_leverage = self.leverage_optimizer.calculate_volatility_adjusted_leverage(
            price_history, opportunity.leverage_recommendation
        )
        opportunity.leverage_recommendation = optimized_leverage
        
        # Recalcular risk/reward com novos targets
        avg_tp = np.mean(opportunity.take_profit_targets)
        risk = abs(opportunity.entry_price - opportunity.stop_loss)
        reward = abs(avg_tp - opportunity.entry_price)
        opportunity.risk_reward_ratio = reward / risk if risk > 0 else 0
        
        return opportunity
    
    async def process_opportunities(self, opportunities: List[TradeOpportunity]) -> List[str]:
        """Processa oportunidades detectadas"""
        registered_opportunities = []
        
        for opportunity in opportunities:
            try:
                # Verificar se já temos muitos trades ativos
                active_count = len(await self.trade_manager.get_active_trades())
                if active_count >= self.trading_settings['max_concurrent_trades']:
                    logger.info(f"Limite de trades ativos atingido ({active_count})")
                    break
                
                # Registrar oportunidade no banco
                opportunity_id = await self.trade_manager.register_opportunity(opportunity)
                registered_opportunities.append(opportunity_id)
                
                # Se auto-trade estiver habilitado, executar automaticamente
                if self.trading_settings['auto_trade_enabled']:
                    await self.auto_execute_trade(opportunity_id, opportunity)
                
            except Exception as e:
                logger.error(f"Erro ao processar oportunidade {opportunity.symbol}: {e}")
        
        return registered_opportunities
    
    async def auto_execute_trade(self, opportunity_id: str, opportunity: TradeOpportunity):
        """Executa um trade automaticamente"""
        try:
            # Calcular tamanho da posição
            position_data = self.risk_calculator.calculate_position_size(
                opportunity.entry_price, \
                opportunity.stop_loss
            )
            
            quantity = position_data['position_size']
            leverage = opportunity.leverage_recommendation
            
            # Executar o trade
            trade_id = await self.trade_manager.execute_trade(
                opportunity_id, quantity, leverage
            )
            
            # Adicionar à gestão de posições
            tp_levels = self.risk_calculator.generate_take_profit_levels(
                opportunity.entry_price,
                opportunity.direction,
                opportunity.stop_loss,
                num_levels=3
            )
            
            self.position_manager.add_position(
                trade_id, \
                opportunity.entry_price, \
                quantity, \
                opportunity.direction, \
                tp_levels
            )
            
            self.active_trades[trade_id] = {
                'opportunity_id': opportunity_id,
                'symbol': opportunity.symbol,
                'entry_time': datetime.now()
            }
            
            logger.info(f"Trade auto-executado: {trade_id}")
            
        except Exception as e:
            logger.error(f"Erro na execução automática: {e}")
    
    async def monitor_active_trades(self, current_prices: Dict[str, float]):
        """Monitora trades ativos para hits de TP e SL"""
        for trade_id in list(self.active_trades.keys()):
            try:
                trade_info = self.active_trades[trade_id]
                symbol = trade_info['symbol']
                
                if symbol not in current_prices:
                    continue
                
                current_price = current_prices[symbol]
                
                # Verificar hits de take profit
                tp_hits = self.position_manager.check_take_profit_hits(trade_id, current_price)
                for tp_hit in tp_hits:
                    await self.trade_manager.send_target_hit_notification(
                        trade_id, tp_hit.level, tp_hit.price, trade_info
                    )
                
                # Verificar stop loss
                await self.trade_manager.check_stop_loss(trade_id, current_price)
                
                # Verificar se o trade foi fechado
                trade_summary = self.position_manager.get_position_summary(trade_id, current_price)
                if trade_summary.get('remaining_quantity', 0) <= 0:
                    # Trade totalmente fechado
                    del self.active_trades[trade_id]
                    logger.info(f"Trade {trade_id} totalmente fechado")
                
            except Exception as e:
                logger.error(f"Erro ao monitorar trade {trade_id}: {e}")
    
    async def get_portfolio_summary(self, current_prices: Dict[str, float]) -> Dict:
        """Retorna resumo do portfólio"""
        total_pnl = 0
        total_unrealized_pnl = 0
        total_realized_pnl = 0
        active_trades_count = len(self.active_trades)
        
        trade_summaries = []
        
        for trade_id in self.active_trades:
            trade_info = self.active_trades[trade_id]
            symbol = trade_info['symbol']
            
            if symbol in current_prices:
                summary = self.position_manager.get_position_summary(
                    trade_id, current_prices[symbol]
                )
                trade_summaries.append(summary)
                
                total_unrealized_pnl += summary.get('unrealized_pnl', 0)
                total_realized_pnl += summary.get('realized_pnl', 0)
        
        total_pnl = total_realized_pnl + total_unrealized_pnl
        total_pnl_percentage = (total_pnl / self.account_balance) * 100
        
        return {
            'account_balance': self.account_balance,
            'total_pnl': total_pnl,
            'total_pnl_percentage': total_pnl_percentage,
            'realized_pnl': total_realized_pnl,
            'unrealized_pnl': total_unrealized_pnl,
            'active_trades_count': active_trades_count,
            'max_trades': self.trading_settings['max_concurrent_trades'],
            'trade_summaries': trade_summaries
        }
    
    async def update_trading_settings(self, new_settings: Dict):
        """Atualiza configurações de trading"""
        self.trading_settings.update(new_settings)
        logger.info(f"Configurações atualizadas: {new_settings}")
    
    async def run_trading_cycle(self, market_data: Dict[str, pd.DataFrame], \
                              current_prices: Dict[str, float]):
        """Executa um ciclo completo de trading"""
        try:
            logger.info("Iniciando ciclo de trading...")
            
            # 1. Detectar oportunidades
            opportunities = await self.analyze_market_and_detect_opportunities(market_data)
            logger.info(f"Detectadas {len(opportunities)} oportunidades")
            
            # 2. Processar oportunidades
            if opportunities:
                registered = await self.process_opportunities(opportunities)
                logger.info(f"Registradas {len(registered)} oportunidades")
            
            # 3. Monitorar trades ativos
            await self.monitor_active_trades(current_prices)
            
            # 4. Gerar resumo do portfólio
            portfolio = await self.get_portfolio_summary(current_prices)
            logger.info(f"PnL Total: ${portfolio['total_pnl']:.2f} ({portfolio['total_pnl_percentage']:.2f}%)")
            
            return {
                'opportunities_detected': len(opportunities),
                'opportunities_registered': len(registered) if opportunities else 0,
                'active_trades': portfolio['active_trades_count'],
                'portfolio_summary': portfolio
            }
            
        except Exception as e:
            logger.error(f"Erro no ciclo de trading: {e}")
            return None

import pandas as pd

# Implementações simplificadas dos indicadores técnicos
def simple_rsi(prices, period=14):
    """Implementação simples do RSI"""
    deltas = np.diff(prices)
    gains = np.where(deltas > 0, deltas, 0)
    losses = np.where(deltas < 0, -deltas, 0)
    
    avg_gains = np.convolve(gains, np.ones(period)/period, mode='valid')
    avg_losses = np.convolve(losses, np.ones(period)/period, mode='valid')
    
    rs = avg_gains / (avg_losses + 1e-10)  # Evitar divisão por zero
    rsi = 100 - (100 / (1 + rs))
    
    return np.concatenate([np.full(period, 50), rsi])  # Preencher valores iniciais

def simple_sma(prices, period):
    """Implementação simples da média móvel simples"""
    return np.convolve(prices, np.ones(period)/period, mode='same')

def simple_obv(close_prices, volumes):
    """Implementação simples do On-Balance Volume"""
    obv = np.zeros(len(close_prices))
    obv[0] = volumes[0]
    
    for i in range(1, len(close_prices)):
        if close_prices[i] > close_prices[i-1]:
            obv[i] = obv[i-1] + volumes[i]
        elif close_prices[i] < close_prices[i-1]:
            obv[i] = obv[i-1] - volumes[i]
        else:
            obv[i] = obv[i-1]
    
    return obv

def simple_mfi(high, low, close, volume, period=14):
    """Implementação simples do Money Flow Index"""
    typical_price = (high + low + close) / 3
    money_flow = typical_price * volume
    
    mfi_values = []
    for i in range(period, len(close)):
        positive_flow = 0
        negative_flow = 0
        
        for j in range(i-period+1, i+1):
            if j > 0:
                if typical_price[j] > typical_price[j-1]:
                    positive_flow += money_flow[j]
                elif typical_price[j] < typical_price[j-1]:
                    negative_flow += money_flow[j]
        
        if negative_flow == 0:
            mfi = 100
        else:
            money_ratio = positive_flow / negative_flow
            mfi = 100 - (100 / (1 + money_ratio))
        
        mfi_values.append(mfi)
    
    return np.concatenate([np.full(period, 50), mfi_values])

        sma_slow = simple_sma(close_prices, 21)
